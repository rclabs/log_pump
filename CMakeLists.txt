cmake_minimum_required(VERSION 3.16)

if(ESP_PLATFORM)
    idf_component_register(INCLUDE_DIRS include REQUIRES freertos)
    target_compile_features(${COMPONENT_LIB} INTERFACE cxx_std_17)
else()
    project(log_pump VERSION 0.1.0 LANGUAGES CXX)
    add_library(log_pump INTERFACE)
    target_include_directories(log_pump INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_compile_features(log_pump INTERFACE cxx_std_17)
    include(CTest)
    if(BUILD_TESTING)
        find_package(Threads REQUIRED)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_executable(recorder_tests tests/recorder_tests.cpp)
        target_link_libraries(recorder_tests PRIVATE log_pump Threads::Threads)
        add_test(NAME recorder_tests COMMAND recorder_tests)
        add_executable(freshness_tests tests/freshness_tests.cpp)
        target_link_libraries(freshness_tests PRIVATE log_pump)
        add_test(NAME freshness_tests COMMAND freshness_tests)
        add_test(NAME tool_tests COMMAND ${Python3_EXECUTABLE} -m unittest discover
            -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_*.py -v)
    endif()
endif()
